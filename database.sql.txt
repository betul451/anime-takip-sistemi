-- =====================================================
-- Anime Takip Sistemi (AniList Güncel Şeması)
-- 11 tablo + 1 trigger + 1 view + 2 procedure
-- Veri Tabanı Yönetimi Dersi Proje Ödevi
-- =====================================================

CREATE DATABASE AnimeTakipDB;
GO
USE AnimeTakipDB;
GO

-- ---------- TABLOLAR ----------

-- Animenin yapımcı stüdyo bilgilerini tutan tablo
CREATE TABLE Studyo (
    StudyoID INT PRIMARY KEY,                 -- AniList stüdyo ID
    Ad NVARCHAR(150) NOT NULL UNIQUE
);

-- Anime türlerini tutan katman tablosu (Aksiyon, Dram vb.)
CREATE TABLE Tur (
    TurID INT PRIMARY KEY,                    -- Sistem içi otomatik verilen tür ID
    Ad NVARCHAR(100) NOT NULL UNIQUE
);

-- Ana anime kataloğu tablosu
CREATE TABLE Anime (
    AnimeID INT PRIMARY KEY,                  -- AniList Media ID
    Baslik NVARCHAR(300) NOT NULL,
    BaslikJaponca NVARCHAR(300) NULL,
    Format NVARCHAR(30) NULL,                 -- TV, Movie, OVA...
    BolumSayisi INT NULL,
    Durum NVARCHAR(50) NULL,                  -- Finished, Releasing vb.
    YayinBaslangic DATE NULL,
    Ozet NVARCHAR(MAX) NULL,
    MalPuani DECIMAL(4,2) NULL,               -- Kaynak API puanı (0-10 arası)
    PosterURL NVARCHAR(500) NULL,
    StudyoID INT NULL REFERENCES Studyo(StudyoID),
    OrtalamaPuan DECIMAL(4,2) NULL,           -- Kullanıcı puanlarından trigger ile güncellenir
    PuanSayisi INT NOT NULL DEFAULT 0         -- Toplam verilen oy sayısı (trigger günceller)
);

-- Anime ile Tur arasındaki çoka-çok (N:M) ilişkiyi çözen köprü tablo
CREATE TABLE Anime_Tur (
    AnimeID INT NOT NULL REFERENCES Anime(AnimeID) ON DELETE CASCADE,
    TurID INT NOT NULL REFERENCES Tur(TurID),
    PRIMARY KEY (AnimeID, TurID)
);

-- AniList etiket verilerini tutan tablo (örn: Time Travel, High School)
CREATE TABLE Etiket (
    EtiketID INT PRIMARY KEY,                 -- AniList Tag ID
    Ad NVARCHAR(100) NOT NULL UNIQUE,
    Kategori NVARCHAR(100) NULL               -- Theme, Setting, Cast vb.
);

-- Anime ile Etiket arasındaki çoka-çok ilişki ve uygunluk oyu tablosu
CREATE TABLE Anime_Etiket (
    AnimeID INT NOT NULL REFERENCES Anime(AnimeID) ON DELETE CASCADE,
    EtiketID INT NOT NULL REFERENCES Etiket(EtiketID) ON DELETE CASCADE,
    Oy INT NULL,                               -- Etiketin animeye uygunluk yüzdesi/oyu
    PRIMARY KEY (AnimeID, EtiketID)
);

-- Animelerdeki karakter bilgilerini tutan tablo
CREATE TABLE Karakter (
    KarakterID INT PRIMARY KEY,               -- AniList Character ID
    Ad NVARCHAR(200) NOT NULL,
    ResimURL NVARCHAR(500) NULL
);

-- Anime ile Karakter arasındaki çoka-çok ilişki ve rol bilgisi tablosu
CREATE TABLE Anime_Karakter (
    AnimeID INT NOT NULL REFERENCES Anime(AnimeID) ON DELETE CASCADE,
    KarakterID INT NOT NULL REFERENCES Karakter(KarakterID),
    Rol NVARCHAR(30) NULL,                    -- Main / Supporting
    PRIMARY KEY (AnimeID, KarakterID)
);

-- Sisteme üye olan kullanıcı bilgilerini tutan tablo
CREATE TABLE Kullanici (
    KullaniciID INT IDENTITY(1,1) PRIMARY KEY,
    KullaniciAdi NVARCHAR(50) NOT NULL UNIQUE,
    Eposta NVARCHAR(150) NOT NULL UNIQUE,
    SifreHash NVARCHAR(200) NOT NULL,
    KayitTarihi DATETIME NOT NULL DEFAULT GETDATE()
);

-- Kullanıcıların kişisel izleme listelerini tutan tablo
CREATE TABLE Izleme_Listesi (
    KullaniciID INT NOT NULL REFERENCES Kullanici(KullaniciID) ON DELETE CASCADE,
    AnimeID INT NOT NULL REFERENCES Anime(AnimeID) ON DELETE CASCADE,
    Durum NVARCHAR(20) NOT NULL
        CHECK (Durum IN (N'Izliyorum', N'Bitirdim', N'Izlenecek', N'Biraktim')),
    IzlenenBolum INT NOT NULL DEFAULT 0,
    EklenmeTarihi DATETIME NOT NULL DEFAULT GETDATE(),
    PRIMARY KEY (KullaniciID, AnimeID)
);

-- Kullanıcıların animelere verdiği puan ve yorumları tutan tablo
CREATE TABLE Puan_Yorum (
    PuanYorumID INT IDENTITY(1,1) PRIMARY KEY,
    KullaniciID INT NOT NULL REFERENCES Kullanici(KullaniciID) ON DELETE CASCADE,
    AnimeID INT NOT NULL REFERENCES Anime(AnimeID) ON DELETE CASCADE,
    Puan INT NOT NULL CHECK (Puan BETWEEN 1 AND 10),
    Yorum NVARCHAR(1000) NULL,
    Tarih DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT UQ_PuanYorum UNIQUE (KullaniciID, AnimeID)   -- Bir kullanıcı bir animeye tek puan verebilir
);
GO

-- ---------- TRIGGER ----------
-- Yeni puan eklenince/güncellenince/silinince Anime tablosundaki ortalama puanı otomatik hesaplar

CREATE TRIGGER trg_PuanGuncelle
ON Puan_Yorum
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE a
    SET OrtalamaPuan = (SELECT AVG(CAST(p.Puan AS DECIMAL(4,2)))
                        FROM Puan_Yorum p WHERE p.AnimeID = a.AnimeID),
        PuanSayisi   = (SELECT COUNT(*)
                        FROM Puan_Yorum p WHERE p.AnimeID = a.AnimeID)
    FROM Anime a
    WHERE a.AnimeID IN (SELECT AnimeID FROM inserted
                        UNION
                        SELECT AnimeID FROM deleted);
END;
GO

-- ---------- VIEW ----------
-- Kullanıcı puanlarına göre en yüksek ortalamaya sahip 50 animeyi stüdyosuyla getiren görünüm

CREATE VIEW vw_EnYuksekPuanlilar AS
SELECT TOP 50
    a.AnimeID, a.Baslik, s.Ad AS Studyo, a.OrtalamaPuan, a.PuanSayisi
FROM Anime a
LEFT JOIN Studyo s ON s.StudyoID = a.StudyoID
WHERE a.PuanSayisi > 0
ORDER BY a.OrtalamaPuan DESC, a.PuanSayisi DESC;
GO

-- ---------- PROCEDURE 1 ----------
-- Kullanıcının izleme listesine anime ekler; zaten listedeyse durumunu günceller

CREATE PROCEDURE sp_ListeyeEkle
    @KullaniciID INT,
    @AnimeID INT,
    @Durum NVARCHAR(20)
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (SELECT 1 FROM Izleme_Listesi
               WHERE KullaniciID = @KullaniciID AND AnimeID = @AnimeID)
        UPDATE Izleme_Listesi
        SET Durum = @Durum
        WHERE KullaniciID = @KullaniciID AND AnimeID = @AnimeID;
    ELSE
        INSERT INTO Izleme_Listesi (KullaniciID, AnimeID, Durum)
        VALUES (@KullaniciID, @AnimeID, @Durum);
END;
GO

-- ---------- PROCEDURE 2 ----------
-- Kullanıcının 8+ puan verdiği türlere göre listesinde olmayan ilk 10 animeyi önerir

CREATE PROCEDURE sp_OneriGetir
    @KullaniciID INT
AS
BEGIN
    SET NOCOUNT ON;

    SELECT TOP 10 a.AnimeID, a.Baslik, a.MalPuani
    FROM Anime a
    WHERE a.AnimeID NOT IN (SELECT AnimeID FROM Izleme_Listesi
                            WHERE KullaniciID = @KullaniciID)
      AND EXISTS (
            SELECT 1
            FROM Anime_Tur atur
            WHERE atur.AnimeID = a.AnimeID
              AND atur.TurID IN (
                    SELECT at2.TurID
                    FROM Puan_Yorum p
                    JOIN Anime_Tur at2 ON at2.AnimeID = p.AnimeID
                    WHERE p.KullaniciID = @KullaniciID AND p.Puan >= 8))
    ORDER BY a.MalPuani DESC;
END;
GO
